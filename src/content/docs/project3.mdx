import Steps from '../../components/Steps.astro';

Project 1에서는 우리가 직접 만든 데이터(사진)로 AI를 다뤘습니다. 이번엔 반대로 **세상에 이미 있는 데이터**를 찾아서 읽고 해석하고, AI와 함께 문제를 해결합니다. 데이터 개념을 먼저 길게 설명하지 않습니다 — 직접 파일을 열어본 뒤에 필요한 개념을 배웁니다.

<Callout kind="mission">
**청소년 또는 우리 사회와 관련된 공공데이터에서 문제의 근거를 찾고, AI와 인간의 역할을 구분한 해결 서비스를 기획하라.**
</Callout>

<Steps>
	<Step title="관심 문제 정하기">청소년 또는 우리 사회와 관련해서 궁금하거나 문제라고 느끼는 주제를 정합니다.</Step>
	<Step title="공공데이터 찾기">
		[공공데이터포털](https://www.data.go.kr), [국가통계포털(KOSIS)](https://kosis.kr) 등에서 관련 데이터를 검색해 다운로드합니다.

		<Callout kind="concept" title="데이터·표본">
		**데이터**는 관찰하거나 측정해서 모아놓은 값들의 모음입니다. 공공데이터는 보통 전체가 아니라 일부를 조사한 **표본**인 경우가 많습니다. 표본이 전체를 얼마나 잘 대표하는지도 함께 확인해야 합니다.
		</Callout>
	</Step>
	<Step title="누가, 언제, 왜 수집했는지 확인">데이터를 만든 기관, 조사 시기, 조사 목적을 확인합니다. 오래된 데이터이거나 특정 목적을 위해 만들어진 데이터일 수 있습니다.</Step>
	<Step title="행과 열의 의미 파악">
		엑셀(또는 구글 시트)로 파일을 열어 표의 구조를 확인합니다.

		<Callout kind="concept" title="행과 열">
		<strong>열(column)</strong>은 데이터가 가진 각각의 항목(**속성/변수**)입니다. (예: 나이, 성별, 하루 스마트폰 사용 시간) <strong>행(row)</strong>은 한 사람 또는 한 사례에 대한 데이터 전체, 표의 한 줄입니다.
		</Callout>
	</Step>
	<Step title="변수와 단위 확인">각 열이 정확히 무엇을 의미하는지, 단위는 무엇인지(%, 명, 시간 등) 확인합니다.</Step>
	<Step title="결측치·이상치·중복 확인">
		<Callout kind="concept" title="결측치·이상치">
		**결측치**는 값이 비어 있는 칸입니다. **이상치**는 다른 값들과 비교했을 때 유난히 튀는 값입니다. (예: 하루 스마트폰 사용 시간이 25시간으로 적힌 경우) 이를 그대로 두고 분석하면 잘못된 결론에 이를 수 있습니다.
		</Callout>
	</Step>
	<Step title="적절한 그래프 만들기">
		데이터 성격에 맞는 그래프(막대, 꺾은선, 산점도 등)를 그립니다.

		<Callout kind="concept" title="데이터 시각화">
		숫자를 그래프나 표로 바꿔서 패턴을 한눈에 보이게 만드는 것을 **시각화**라고 합니다. 같은 데이터라도 **그래프 종류, 축의 범위, 비교 대상, 집계 방식**에 따라 인상이 크게 달라질 수 있습니다.
		</Callout>
	</Step>
	<Step title="패턴 해석">
		<Callout kind="discuss">
		모둠원 각자 같은 그래프를 보고 해석을 한 문장씩 적은 뒤 비교하세요. 같은 데이터를 보고도 해석이 다르다면, 왜 다를까요?
		</Callout>

		<Callout kind="concept" title="상관관계와 인과관계">
		**상관관계**는 두 변수가 함께 움직이는 경향입니다. 다만 상관관계가 있다고 해서 한쪽이 다른 쪽의 **원인**이라고 단정할 수는 없습니다 — 이 둘을 구분하는 것이 데이터를 비판적으로 읽는 핵심입니다.
		</Callout>
	</Step>
	<Step title="AI에게 분석 요청">정리한 데이터를 AI에게 주고 "이 데이터를 분석해줘"라고 요청합니다.</Step>
	<Step title="AI 분석을 원자료와 계산으로 검증">
		<Callout kind="caution" title="AI 분석, 그대로 믿지 않기">
		- AI가 열 이름과 단위를 제대로 이해했는가?
		- 존재하지 않는 값을 만들지 않았는가?
		- 계산 결과를 내가 직접 재현할 수 있는가?
		- 상관관계를 인과관계처럼 말하지 않았는가?
		- 데이터에 없는 집단까지 일반화하지 않았는가?
		- 반례와 한계를 빠뜨리지 않았는가?
		</Callout>
	</Step>
	<Step title="데이터 근거로 주장 작성">
		<Callout kind="concept" title="근거와 주장">
		"청소년의 스마트폰 사용 시간이 늘고 있다"는 주장을 하려면, 근거가 되는 구체적인 데이터(몇 %가, 몇 년 사이에, 얼마나)를 함께 제시해야 설득력이 생깁니다.
		</Callout>
	</Step>
	<Step title="AI 서비스 기획">이 문제 중 AI가 도울 수 있는 부분이 무엇인지 정하고, 한 문단짜리 서비스 기획안을 작성합니다.</Step>
	<Step title="AI와 인간의 역할 구분">AI가 할 일과 사람이 책임져야 할 일을 표로 정리합니다. (예: AI는 패턴을 찾아 알려주고, 최종 판단과 실행은 사람이 한다.)</Step>
	<Step title="한계와 위험 발표">이 서비스가 틀릴 수 있는 상황, 조심해야 할 위험을 함께 발표합니다.</Step>
</Steps>

<Callout kind="exit">
오늘 다룬 데이터에서 상관관계는 있지만 인과관계라고 말할 수 없는 예를 하나 찾아 적어보세요.
</Callout>
